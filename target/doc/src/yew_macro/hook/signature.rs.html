38pub struct HookSignature {
39    pub hook_lifetime: Lifetime,
40    pub sig: Signature,
41    pub output_type: Type,
42    pub needs_boxing: bool,
43}
44
45impl HookSignature {
46    fn rewrite_return_type(hook_lifetime: &Lifetime, rt_type: &ReturnType) -> (ReturnType, Type) {
47        let bound = quote! { #hook_lifetime + };
48
49        match rt_type {
50            ReturnType::Default => (
51                parse_quote! { -> impl #bound ::yew::functional::Hook<Output = ()> },
52                parse_quote! { () },
53            ),
54            ReturnType::Type(arrow, ref return_type) => {
55                if let Type::Reference(ref m) = &**return_type {
56                    if m.lifetime.is_none() {
57                        let mut return_type_ref = m.clone();
58                        return_type_ref.lifetime = parse_quote!('hook);
59
60                        let return_type_ref = Type::Reference(return_type_ref);
61
62                        return (
63                            parse_quote_spanned! {
64                                return_type.span() => #arrow impl #bound ::yew::functional::Hook<Output = #return_type_ref>
65                            },
66                            return_type_ref,
67                        );
68                    }
69                }
70
71                (
72                    parse_quote_spanned! {
73                        return_type.span() => #arrow impl #bound ::yew::functional::Hook<Output = #return_type>
74                    },
75                    *return_type.clone(),
76                )
77            }
78        }
79    }
80
81    /// Rewrites a Hook Signature and extracts information.
82    pub fn rewrite(sig: &Signature) -> Self {
83        let mut sig = sig.clone();
84
85        let mut arg_info = CollectArgs::new();
86        arg_info.visit_signature_mut(&mut sig);
87
88        let mut lifetimes = lifetime::CollectLifetimes::new("'arg", sig.ident.span());
89        for arg in sig.inputs.iter_mut() {
90            match arg {
91                FnArg::Receiver(arg) => lifetimes.visit_receiver_mut(arg),
92                FnArg::Typed(arg) => lifetimes.visit_type_mut(&mut arg.ty),
93            }
94        }
95
96        let Signature {
97            ref mut generics,
98            output: ref return_type,
99            ..
100        } = sig;
101
102        let hook_lifetime = {
103            let hook_lifetime = Lifetime::new("'hook", Span::mixed_site());
104            generics.params = {
105                let elided_lifetimes = &lifetimes.elided;
106                let params = &generics.params;
107
108                parse_quote!(#hook_lifetime, #(#elided_lifetimes,)* #params)
109            };
110
111            let mut where_clause = generics
112                .where_clause
113                .clone()
114                .unwrap_or_else(|| WhereClause {
115                    where_token: token::Where {
116                        span: Span::mixed_site(),
117                    },
118                    predicates: Default::default(),
119                });
120
121            for elided in lifetimes.elided.iter() {
122                where_clause
123                    .predicates
124                    .push(parse_quote!(#elided: #hook_lifetime));
125            }
126
127            for explicit in lifetimes.explicit.iter() {
128                where_clause
129                    .predicates
130                    .push(parse_quote!(#explicit: #hook_lifetime));
131            }
132
133            for type_param in generics.type_params() {
134                let type_param_ident = &type_param.ident;
135                where_clause
136                    .predicates
137                    .push(parse_quote!(#type_param_ident: #hook_lifetime));
138            }
139
140            generics.where_clause = Some(where_clause);
141
142            hook_lifetime
143        };
144
145        let (output, output_type) = Self::rewrite_return_type(&hook_lifetime, return_type);
146        sig.output = output;
147
148        Self {
149            hook_lifetime,
150            sig,
151            output_type,
152            needs_boxing: arg_info.needs_boxing,
153        }
154    }

156    pub fn phantom_types(&self) -> Vec<Ident> {

172    pub fn input_args(&self) -> Vec<Ident> {
173        self.sig
174            .inputs
175            .iter()
176            .filter_map(|m| {
177                if let FnArg::Typed(m) = m {
178                    if let Pat::Ident(ref m) = *m.pat {
179                        return Some(m.ident.clone());
180                    }
181                }
182
183                None
184            })
185            .collect()
186    }
187
188    pub fn input_types(&self) -> Vec<Type> {
189        self.sig
190            .inputs
191            .iter()
192            .filter_map(|m| {
193                if let FnArg::Typed(m) = m {
194                    return Some(*m.ty.clone());
195                }
196
197                None
198            })
199            .collect()
200    }

202    pub fn call_generics(&self) -> TokenStream {
203        let mut generics = self.sig.generics.clone();
204
205        // We need to filter out lifetimes.
206        generics.params = generics
207            .params
208            .into_iter()
209            .filter(|m| !matches!(m, GenericParam::Lifetime(_)))
210            .collect();
211
212        let (_impl_generics, ty_generics, _where_clause) = generics.split_for_impl();
213        ty_generics.as_turbofish().to_token_stream()
214    }
