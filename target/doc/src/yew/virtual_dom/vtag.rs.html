24/// Value field corresponding to an [Element]'s `value` property
25#[derive(Clone, Debug, Eq, PartialEq)]
26pub(crate) struct Value<T>(Option<AttrValue>, PhantomData<T>);
27
28impl<T> Default for Value<T> {
29    fn default() -> Self {
30        Self::new(None)
31    }
32}
33
34impl<T> Value<T> {
35    /// Create a new value. The caller should take care that the value is valid for the element's
36    /// `value` property
37    fn new(value: Option<AttrValue>) -> Self {
38        Value(value, PhantomData)
39    }
40
41    /// Set a new value. The caller should take care that the value is valid for the element's
42    /// `value` property
43    fn set(&mut self, value: Option<AttrValue>) {
44        self.0 = value;
45    }
46}
47
48impl<T> Deref for Value<T> {
49    type Target = Option<AttrValue>;
50
51    fn deref(&self) -> &Self::Target {
52        &self.0
53    }
54}

56/// Fields specific to
57/// [InputElement](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input) [VTag](crate::virtual_dom::VTag)s
58#[derive(Debug, Clone, Default, Eq, PartialEq)]
59pub(crate) struct InputFields {
60    /// Contains a value of an
61    /// [InputElement](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input).
62    pub(crate) value: Value<InputElement>,
63    /// Represents `checked` attribute of
64    /// [input](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input#attr-checked).
65    /// It exists to override standard behavior of `checked` attribute, because
66    /// in original HTML it sets `defaultChecked` value of `InputElement`, but for reactive
67    /// frameworks it's more useful to control `checked` value of an `InputElement`.
68    pub(crate) checked: Option<bool>,
69}
70
71impl Deref for InputFields {
72    type Target = Value<InputElement>;
73
74    fn deref(&self) -> &Self::Target {
75        &self.value
76    }
77}
78
79impl DerefMut for InputFields {
80    fn deref_mut(&mut self) -> &mut Self::Target {
81        &mut self.value
82    }
83}

87    fn new(value: Option<AttrValue>, checked: Option<bool>) -> Self {
88        Self {
89            value: Value::new(value),
90            checked,
91        }
92    }

95/// [VTag] fields that are specific to different [VTag] kinds.
96/// Decreases the memory footprint of [VTag] by avoiding impossible field and value combinations.
97#[derive(Debug, Clone)]
98pub(crate) enum VTagInner {
99    /// Fields specific to
100    /// [InputElement](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input)
101    /// [VTag]s
102    Input(InputFields),
103    /// Fields specific to
104    /// [TextArea](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/textarea)
105    /// [VTag]s
106    Textarea {
107        /// Contains a value of an
108        /// [TextArea](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/textarea)
109        value: Value<TextAreaElement>,
110    },
111    /// Fields for all other kinds of [VTag]s
112    Other {
113        /// A tag of the element.
114        tag: Cow<'static, str>,
115        /// children of the element.
116        children: VNode,
117    },
118}

120/// A type for a virtual
121/// [Element](https://developer.mozilla.org/en-US/docs/Web/API/Element)
122/// representation.
123#[derive(Debug, Clone)]
124pub struct VTag {
125    /// [VTag] fields that are specific to different [VTag] kinds.
126    pub(crate) inner: VTagInner,
127    /// List of attached listeners.
128    pub(crate) listeners: Listeners,
129    /// A node reference used for DOM access in Component lifecycle methods
130    pub node_ref: NodeRef,
131    /// List of attributes.
132    pub attributes: Attributes,
133    pub key: Option<Key>,
134}

136impl VTag {
137    /// Creates a new [VTag] instance with `tag` name (cannot be changed later in DOM).
138    pub fn new(tag: impl Into<Cow<'static, str>>) -> Self {
139        let tag: Cow<'static, str> = tag.into();
140        Self::new_base(
141            match &*tag.to_ascii_lowercase() {
142                "input" => VTagInner::Input(Default::default()),
143                "textarea" => VTagInner::Textarea {
144                    value: Default::default(),
145                },
146                _ => VTagInner::Other {
147                    tag,
148                    children: Default::default(),
149                },
150            },
151            Default::default(),
152            Default::default(),
153            Default::default(),
154            Default::default(),
155        )
156    }
157
158    /// Creates a new
159    /// [InputElement](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input) [VTag]
160    /// instance.
161    ///
162    /// Unlike [VTag::new()], this sets all the public fields of [VTag] in one call. This allows the
163    /// compiler to inline property and child list construction in the `html!` macro. This enables
164    /// higher instruction parallelism by reducing data dependency and avoids `memcpy` of Vtag
165    /// fields.
166    #[doc(hidden)]
167    #[allow(clippy::too_many_arguments)]
168    pub fn __new_input(
169        value: Option<AttrValue>,
170        checked: Option<bool>,
171        node_ref: NodeRef,
172        key: Option<Key>,
173        // at bottom for more readable macro-expanded coded
174        attributes: Attributes,
175        listeners: Listeners,
176    ) -> Self {
177        VTag::new_base(
178            VTagInner::Input(InputFields::new(
179                value,
180                // In HTML node `checked` attribute sets `defaultChecked` parameter,
181                // but we use own field to control real `checked` parameter
182                checked,
183            )),
184            node_ref,
185            key,
186            attributes,
187            listeners,
188        )
189    }

192    /// [TextArea](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/textarea) [VTag]
193    /// instance.
194    ///
195    /// Unlike [VTag::new()], this sets all the public fields of [VTag] in one call. This allows the
196    /// compiler to inline property and child list construction in the `html!` macro. This enables
197    /// higher instruction parallelism by reducing data dependency and avoids `memcpy` of Vtag
198    /// fields.

406    /// Add event listener on the [VTag]'s  [Element](web_sys::Element).
407    /// Returns `true` if the listener has been added, `false` otherwise.
408    pub fn add_listener(&mut self, listener: Rc<dyn Listener>) -> bool {
409        match &mut self.listeners {
410            Listeners::None => {
411                self.set_listeners([Some(listener)].into());
412                true
413            }
414            Listeners::Pending(listeners) => {
415                let mut listeners = mem::take(listeners).into_vec();
416                listeners.push(Some(listener));
417
418                self.set_listeners(listeners.into());
419                true
420            }
421        }
422    }
423
424    /// Set event listeners on the [VTag]'s  [Element](web_sys::Element)
425    pub fn set_listeners(&mut self, listeners: Box<[Option<Rc<dyn Listener>>]>) {
426        self.listeners = Listeners::Pending(listeners);
427    }

430impl PartialEq for VTag {
431    fn eq(&self, other: &VTag) -> bool {
432        use VTagInner::*;
433
434        (match (&self.inner, &other.inner) {
435            (Input(l), Input(r)) => l == r,
436            (Textarea { value: value_l }, Textarea { value: value_r }) => value_l == value_r,
437            (Other { tag: tag_l, .. }, Other { tag: tag_r, .. }) => tag_l == tag_r,
438            _ => false,
439        }) && self.listeners.eq(&other.listeners)
440            && self.attributes == other.attributes
441            // Diff children last, as recursion is the most expensive
442            && match (&self.inner, &other.inner) {
443                (Other { children: ch_l, .. }, Other { children: ch_r, .. }) => ch_l == ch_r,
444                _ => true,
445            }
446    }
447}

449#[cfg(feature = "ssr")]
450mod feat_ssr {
451    use std::fmt::Write;
452
453    use super::*;
454    use crate::feat_ssr::VTagKind;
455    use crate::html::AnyScope;
456    use crate::platform::fmt::BufWriter;
457    use crate::virtual_dom::VText;
458
459    // Elements that cannot have any child elements.
460    static VOID_ELEMENTS: &[&str; 14] = &[
461        "area", "base", "br", "col", "embed", "hr", "img", "input", "link", "meta", "param",
462        "source", "track", "wbr",
463    ];
464
465    impl VTag {
466        pub(crate) async fn render_into_stream(
467            &self,
468            w: &mut BufWriter,
469            parent_scope: &AnyScope,
470            hydratable: bool,
471        ) {
472            let _ = w.write_str("<");
473            let _ = w.write_str(self.tag());
474
475            let write_attr = |w: &mut BufWriter, name: &str, val: Option<&str>| {
476                let _ = w.write_str(" ");
477                let _ = w.write_str(name);
478
479                if let Some(m) = val {
480                    let _ = w.write_str("=\"");
481                    let _ = w.write_str(&html_escape::encode_double_quoted_attribute(m));
482                    let _ = w.write_str("\"");
483                }
484            };
485
486            if let VTagInner::Input(_) = self.inner {
487                if let Some(m) = self.value() {
488                    write_attr(w, "value", Some(m));
489                }
490
491                // Setting is as an attribute sets the `defaultChecked` property. Only emit this
492                // if it's explicitly set to checked.
493                if self.checked() == Some(true) {
494                    write_attr(w, "checked", None);
495                }
496            }
497
498            for (k, v) in self.attributes.iter() {
499                write_attr(w, k, Some(v));
500            }
501
502            let _ = w.write_str(">");
503
504            match self.inner {
505                VTagInner::Input(_) => {}
506                VTagInner::Textarea { .. } => {
507                    if let Some(m) = self.value() {
508                        VText::new(m.to_owned())
509                            .render_into_stream(w, parent_scope, hydratable, VTagKind::Other)
510                            .await;
511                    }
512
513                    let _ = w.write_str("</textarea>");
514                }
515                VTagInner::Other {
516                    ref tag,
517                    ref children,
518                    ..
519                } => {
520                    if !VOID_ELEMENTS.contains(&tag.as_ref()) {
521                        children
522                            .render_into_stream(w, parent_scope, hydratable, tag.into())
523                            .await;
524
525                        let _ = w.write_str("</");
526                        let _ = w.write_str(tag);
527                        let _ = w.write_str(">");
528                    } else {
529                        // We don't write children of void elements nor closing tags.
530                        debug_assert!(
531                            match children {
532                                VNode::VList(m) => m.is_empty(),
533                                _ => false,
534                            },
535                            "{tag} cannot have any children!"
536                        );
537                    }
538                }
539            }
540        }
541    }
542}
