3use std::cell::RefCell;
4use std::rc::Rc;
5
6use web_sys::{Element, Node};
7
8/// A position in the list of children of an implicit parent [`Element`].
9///
10/// This can either be in front of a `DomSlot::at(next_sibling)`, at the end of the list with
11/// `DomSlot::at_end()`, or a dynamic position in the list with [`DynamicDomSlot::to_position`].
12#[derive(Clone)]
13pub(crate) struct DomSlot {
14    variant: DomSlotVariant,
15}
16
17#[derive(Clone)]
18enum DomSlotVariant {
19    Node(Option<Node>),
20    Chained(DynamicDomSlot),
21}
22
23/// A dynamic dom slot can be reassigned. This change is also seen by the [`DomSlot`] from
24/// [`Self::to_position`] before the reassignment took place.
25#[derive(Clone)]
26pub(crate) struct DynamicDomSlot {
27    target: Rc<RefCell<DomSlot>>,
28}
29
30impl std::fmt::Debug for DomSlot {
31    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
32        self.with_next_sibling(|n| {
33            write!(
34                f,
35                "DomSlot {{ next_sibling: {:?} }}",
36                n.map(crate::utils::print_node)
37            )
38        })
39    }
40}
41
42impl std::fmt::Debug for DynamicDomSlot {
43    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
44        write!(f, "{:#?}", *self.target.borrow())
45    }
46}

48#[cfg(debug_assertions)]
49thread_local! {
50    // A special marker element that should not be referenced
51    static TRAP: Node = gloo::utils::document().create_element("div").unwrap().into();
52}
53
54impl DomSlot {
55    /// Denotes the position just before the given node in its parent's list of children.
56    pub fn at(next_sibling: Node) -> Self {
57        Self::create(Some(next_sibling))
58    }
59
60    /// Denotes the position at the end of a list of children. The parent is implicit.
61    pub fn at_end() -> Self {
62        Self::create(None)
63    }
64
65    pub fn create(next_sibling: Option<Node>) -> Self {
66        Self {
67            variant: DomSlotVariant::Node(next_sibling),
68        }
69    }
70
71    /// A new "placeholder" [DomSlot] that should not be used to insert nodes
72    #[inline]
73    pub fn new_debug_trapped() -> Self {
74        #[cfg(debug_assertions)]
75        {
76            Self::at(TRAP.with(|trap| trap.clone()))
77        }
78        #[cfg(not(debug_assertions))]
79        {
80            Self::at_end()
81        }
82    }
83
84    /// Get the [Node] that comes just after the position, or `None` if this denotes the position at
85    /// the end
86    fn with_next_sibling<R>(&self, f: impl FnOnce(Option<&Node>) -> R) -> R {
87        let checkedf = |node: Option<&Node>| {
88            #[cfg(debug_assertions)]
89            TRAP.with(|trap| {
90                assert!(
91                    node != Some(trap),
92                    "Should not use a trapped DomSlot. Please report this as an internal bug in \
93                     yew."
94                )
95            });
96            f(node)
97        };
98
99        match &self.variant {
100            DomSlotVariant::Node(ref n) => checkedf(n.as_ref()),
101            DomSlotVariant::Chained(ref chain) => chain.with_next_sibling(checkedf),
102        }
103    }
104
105    /// Insert a [Node] at the position denoted by this slot. `parent` must be the actual parent
106    /// element of the children that this slot is implicitly a part of.
107    pub(super) fn insert(&self, parent: &Element, node: &Node) {
108        self.with_next_sibling(|next_sibling| {
109            parent
110                .insert_before(node, next_sibling)
111                .unwrap_or_else(|err| {
112                    let msg = if next_sibling.is_some() {
113                        "failed to insert node before next sibling"
114                    } else {
115                        "failed to append child"
116                    };
117                    // Log normally, so we can inspect the nodes in console
118                    gloo::console::error!(msg, err, parent, next_sibling, node);
119                    // Log via tracing for consistency
120                    tracing::error!(msg);
121                    // Panic to short-curcuit and fail
122                    panic!("{}", msg)
123                });
124        });
125    }
126
127    #[cfg(target_arch = "wasm32")]
128    #[cfg(test)]
129    fn get(&self) -> Option<Node> {
130        self.with_next_sibling(|n| n.cloned())
131    }
132}
133
134impl DynamicDomSlot {
135    /// Create a dynamic dom slot that initially represents ("targets") the same slot as the
136    /// argument.
137    pub fn new(initial_position: DomSlot) -> Self {
138        Self {
139            target: Rc::new(RefCell::new(initial_position)),
140        }
141    }
142
143    pub fn new_debug_trapped() -> Self {
144        Self::new(DomSlot::new_debug_trapped())
145    }
146
147    /// Change the [`DomSlot`] that is targeted. Subsequently, this will behave as if `self` was
148    /// created from the passed DomSlot in the first place.
149    pub fn reassign(&self, next_position: DomSlot) {
150        // TODO: is not defensive against accidental reference loops
151        *self.target.borrow_mut() = next_position;
152    }
153
154    /// Get a [`DomSlot`] that gets automatically updated when `self` gets reassigned. All such
155    /// slots are equivalent to each other and point to the same position.
156    pub fn to_position(&self) -> DomSlot {
157        DomSlot {
158            variant: DomSlotVariant::Chained(self.clone()),
159        }
160    }
161
162    fn with_next_sibling<R>(&self, f: impl FnOnce(Option<&Node>) -> R) -> R {
163        // we use an iterative approach to traverse a possible long chain for references
164        // see for example issue #3043 why a recursive call is impossible for large lists in vdom
165
166        // TODO: there could be some data structure that performs better here. E.g. a balanced tree
167        // with parent pointers come to mind, but they are a bit fiddly to implement in rust
168        let mut this = self.target.clone();
169        loop {
170            //                          v------- borrow lives for this match expression
171            let next_this = match &this.borrow().variant {
172                DomSlotVariant::Node(ref n) => break f(n.as_ref()),
173                // We clone an Rc here temporarily, so that we don't have to consume stack
174                // space. The alternative would be to keep the
175                // `Ref<'_, DomSlot>` above in some temporary buffer
176                DomSlotVariant::Chained(ref chain) => chain.target.clone(),
177            };
178            this = next_this;
179        }
180    }
181}

184#[cfg(test)]
185mod layout_tests {
186    use gloo::utils::document;
187    use wasm_bindgen_test::{wasm_bindgen_test as test, wasm_bindgen_test_configure};
188
189    use super::*;
190
191    wasm_bindgen_test_configure!(run_in_browser);
192
193    #[test]
194    fn new_at_and_get() {
195        let node = document().create_element("p").unwrap();
196        let position = DomSlot::at(node.clone().into());
197        assert_eq!(
198            position.get().unwrap(),
199            node.clone().into(),
200            "expected the DomSlot to be at {node:#?}"
201        );
202    }
203
204    #[test]
205    fn new_at_end_and_get() {
206        let position = DomSlot::at_end();
207        assert!(
208            position.get().is_none(),
209            "expected the DomSlot to not have a next sibling"
210        );
211    }
212
213    #[test]
214    fn get_through_dynamic() {
215        let original = DomSlot::at(document().create_element("p").unwrap().into());
216        let target = DynamicDomSlot::new(original.clone());
217        assert_eq!(
218            target.to_position().get(),
219            original.get(),
220            "expected {target:#?} to point to the same position as {original:#?}"
221        );
222    }
223
224    #[test]
225    fn get_after_reassign() {
226        let target = DynamicDomSlot::new(DomSlot::at_end());
227        let target_pos = target.to_position();
228        // We reassign *after* we called `to_position` here to be strict in the test
229        let replacement = DomSlot::at(document().create_element("p").unwrap().into());
230        target.reassign(replacement.clone());
231        assert_eq!(
232            target_pos.get(),
233            replacement.get(),
234            "expected {target:#?} to point to the same position as {replacement:#?}"
235        );
236    }
237
238    #[test]
239    fn get_chain_after_reassign() {
240        let middleman = DynamicDomSlot::new(DomSlot::at_end());
241        let target = DynamicDomSlot::new(middleman.to_position());
242        let target_pos = target.to_position();
243        assert!(
244            target.to_position().get().is_none(),
245            "should not yet point to a node"
246        );
247        // Now reassign the middle man, but get the node from `target`
248        let replacement = DomSlot::at(document().create_element("p").unwrap().into());
249        middleman.reassign(replacement.clone());
250        assert_eq!(
251            target_pos.get(),
252            replacement.get(),
253            "expected {target:#?} to point to the same position as {replacement:#?}"
254        );
255    }
