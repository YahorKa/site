41/// A Bundle.
42///
43/// Each component holds a bundle that represents a realised layout, designated by a [VNode].
44///
45/// This is not to be confused with [BComp], which represents a component in the position of a
46/// bundle layout.
47#[derive(Debug)]
48pub(crate) struct Bundle(BNode);
49
50impl Bundle {
51    /// Creates a new bundle.
52
53    pub const fn new() -> Self {
54        Self(BNode::List(BList::new()))
55    }
56
57    /// Shifts the bundle into a different position.
58    pub fn shift(&self, next_parent: &Element, slot: DomSlot) {
59        self.0.shift(next_parent, slot);
60    }
61
62    /// Applies a virtual dom layout to current bundle.
63    pub fn reconcile(
64        &mut self,
65        root: &BSubtree,
66        parent_scope: &AnyScope,
67        parent: &Element,
68        slot: DomSlot,
69        next_node: VNode,
70    ) -> DomSlot {
71        next_node.reconcile_node(root, parent_scope, parent, slot, &mut self.0)
72    }
73
74    /// Detaches current bundle.
75    pub fn detach(self, root: &BSubtree, parent: &Element, parent_to_detach: bool) {
76        self.0.detach(root, parent, parent_to_detach);
77    }
78}
