19/// DOM-Types that capture (bubbling) events. This generally includes event targets,
20/// but also subtree roots.
21pub trait EventGrating {
22    fn subtree_id(&self) -> Option<TreeId>;
23    fn set_subtree_id(&self, tree_id: TreeId);
24    // When caching, we key on the length of the `composed_path`. Important to check
25    // considering event retargeting!
26    fn cache_key(&self) -> Option<u32>;
27    fn set_cache_key(&self, key: u32);
28}

31extern "C" {
32    // Duck-typing, not a real class on js-side. On rust-side, use impls of EventGrating below
33    type EventTargetable;
34    #[wasm_bindgen(method, getter = __yew_subtree_id, structural)]
35    fn subtree_id(this: &EventTargetable) -> Option<TreeId>;
36    #[wasm_bindgen(method, setter = __yew_subtree_id, structural)]
37    fn set_subtree_id(this: &EventTargetable, id: TreeId);
38    #[wasm_bindgen(method, getter = __yew_subtree_cache_key, structural)]
39    fn cache_key(this: &EventTargetable) -> Option<u32>;
40    #[wasm_bindgen(method, setter = __yew_subtree_cache_key, structural)]
41    fn set_cache_key(this: &EventTargetable, key: u32);
42}

44macro_rules! impl_event_grating {
45    ($($t:ty);* $(;)?) => {
46        $(
47            impl EventGrating for $t {
48                fn subtree_id(&self) -> Option<TreeId> {
49                    self.unchecked_ref::<EventTargetable>().subtree_id()
50                }
51                fn set_subtree_id(&self, tree_id: TreeId) {
52                    self.unchecked_ref::<EventTargetable>()
53                        .set_subtree_id(tree_id);
54                }
55                fn cache_key(&self) -> Option<u32> {
56                    self.unchecked_ref::<EventTargetable>().cache_key()
57                }
58                fn set_cache_key(&self, key: u32) {
59                    self.unchecked_ref::<EventTargetable>().set_cache_key(key)
60                }
61            }
62        )*
63    }
64}

71/// The TreeId is the additional payload attached to each listening element
72/// It identifies the host responsible for the target. Events not matching
73/// are ignored during handling
74type TreeId = u32;
75
76/// Special id for caching the fact that some event should not be handled
77static NONE_TREE_ID: TreeId = 0;
78static NEXT_ROOT_ID: AtomicU32 = AtomicU32::new(1);
79
80fn next_root_id() -> TreeId {
81    NEXT_ROOT_ID.fetch_add(1, Ordering::SeqCst)
82}

118// FIXME: this is a reproduction of gloo's EventListener to work around #2989
119// change back to gloo's implementation once it has been decided how to fix this upstream
120// The important part is that we use `Fn` instead of `FnMut` below!
121type EventClosure = Closure<dyn Fn(&Event)>;
122#[derive(Debug)]
123#[must_use = "event listener will never be called after being dropped"]
124struct EventListener {
125    target: HtmlEventTarget,
126    event_type: Cow<'static, str>,
127    callback: Option<EventClosure>,
128}
129
130impl Drop for EventListener {
131    #[inline]
132    fn drop(&mut self) {
133        if let Some(ref callback) = self.callback {
134            self.target
135                .remove_event_listener_with_callback_and_bool(
136                    &self.event_type,
137                    callback.as_ref().unchecked_ref(),
138                    true, // Always capture
139                )
140                .unwrap_throw();
141        }
142    }
143}
144
145impl EventListener {
146    fn new(
147        target: &HtmlEventTarget,
148        desc: &EventDescriptor,
149        callback: impl 'static + Fn(&Event),
150    ) -> Self {
151        let event_type = desc.kind.type_name();
152
153        let callback = Closure::wrap(Box::new(callback) as Box<dyn Fn(&Event)>);
154        // defaults: { once: false }
155        let mut options = AddEventListenerOptions::new();
156        options.capture(true).passive(desc.passive);
157
158        target
159            .add_event_listener_with_callback_and_add_event_listener_options(
160                intern(&event_type),
161                callback.as_ref().unchecked_ref(),
162                &options,
163            )
164            .unwrap_throw();
165
166        EventListener {
167            target: target.clone(),
168            event_type,
169            callback: Some(callback),
170        }
171    }
172
173    #[cfg(not(test))]
174    fn forget(mut self) {
175        if let Some(callback) = self.callback.take() {
176            // Should always match, but no need to introduce a panic path here
177            callback.forget();
178        }
179    }
180}

182/// Ensures event handler registration.
183// Separate struct to DRY, while avoiding partial struct mutability.
184#[derive(Debug)]
185struct HostHandlers {
186    /// The host element where events are registered
187    host: HtmlEventTarget,
188
189    /// Keep track of all listeners to drop them on registry drop.
190    /// The registry is never dropped in production.
191    #[cfg(test)]
192    registered: Vec<(ListenerKind, EventListener)>,
193}
194
195impl HostHandlers {
196    fn new(host: HtmlEventTarget) -> Self {
197        Self {
198            host,
199            #[cfg(test)]
200            registered: Vec::default(),
201        }
202    }
203
204    fn add_listener(&mut self, desc: &EventDescriptor, callback: impl 'static + Fn(&Event)) {
205        let cl = EventListener::new(&self.host, desc, callback);
206
207        // Never drop the closure as this event handler is static
208        #[cfg(not(test))]
209        cl.forget();
210        #[cfg(test)]
211        self.registered.push((desc.kind.clone(), cl));
212    }
213}

248/// Per tree data, shared between all subtrees in the hierarchy
249#[derive(Debug, Default)]
250struct AppData {
251    subtrees: HashSet<WeakSubtree>,
252    listening: HashSet<EventDescriptor>,
253}
254
255impl AppData {
256    fn add_subtree(&mut self, subtree: &Rc<SubtreeData>) {
257        for event in self.listening.iter() {
258            subtree.add_listener(event);
259        }
260        self.subtrees.insert(WeakSubtree {
261            subtree_id: subtree.subtree_id,
262            weak_ref: Rc::downgrade(subtree),
263        });
264    }
265
266    fn ensure_handled(&mut self, desc: &EventDescriptor) {
267        if !self.listening.insert(desc.clone()) {
268            return;
269        }
270        self.subtrees.retain(|subtree| {
271            if let Some(subtree) = subtree.weak_ref.upgrade() {
272                subtree.add_listener(desc);
273                true
274            } else {
275                false
276            }
277        })
278    }
279}

282static BUBBLE_EVENTS: AtomicBool = AtomicBool::new(true);

284/// Set, if events should bubble up the DOM tree, calling any matching callbacks.
285///
286/// Bubbling is enabled by default. Disabling bubbling can lead to substantial improvements in event
287/// handling performance.
288///
289/// This function should be called before any component is mounted.
290#[cfg(feature = "csr")]
291pub fn set_event_bubbling(bubble: bool) {
292    BUBBLE_EVENTS.store(bubble, Ordering::Relaxed);
293}

295struct BrandingSearchResult {
296    branding: TreeId,
297    closest_branded_ancestor: Element,
298}

300fn shadow_aware_parent(el: &Element) -> Option<Element> {
301    match el.parent_element() {
302        s @ Some(_) => s,
303        None => el.parent_node()?.dyn_ref::<ShadowRoot>().map(|h| h.host()),
304    }
305}

307/// Deduce the subtree an element is part of. This already partially starts the bubbling
308/// process, as long as no listeners are encountered.
309/// Subtree roots are always branded with their own subtree id.
310fn find_closest_branded_element(mut el: Element, do_bubble: bool) -> Option<BrandingSearchResult> {
311    if !do_bubble {
312        let branding = el.subtree_id()?;
313        Some(BrandingSearchResult {
314            branding,
315            closest_branded_ancestor: el,
316        })
317    } else {
318        let responsible_tree_id = loop {
319            if let Some(tree_id) = el.subtree_id() {
320                break tree_id;
321            }
322            el = shadow_aware_parent(&el)?;
323        };
324        Some(BrandingSearchResult {
325            branding: responsible_tree_id,
326            closest_branded_ancestor: el,
327        })
328    }
329}

331/// Iterate over all potentially listening elements in bubbling order.
332/// If bubbling is turned off, yields at most a single element.
333fn start_bubbling_from(
334    subtree: &SubtreeData,
335    root_or_listener: Element,
336    should_bubble: bool,
337) -> impl '_ + Iterator<Item = (&'_ SubtreeData, Element)> {
338    let start = subtree.bubble_to_inner_element(root_or_listener, should_bubble);
339
340    std::iter::successors(start, move |(subtree, element)| {
341        if !should_bubble {
342            return None;
343        }
344        let parent = shadow_aware_parent(element)?;
345        subtree.bubble_to_inner_element(parent, true)
346    })
347}

349impl SubtreeData {
350    fn new_ref(host_element: &HtmlEventTarget, parent: Option<ParentingInformation>) -> Rc<Self> {
351        let tree_root_id = next_root_id();
352        let event_registry = Registry::new();
353        let host_handlers = HostHandlers::new(host_element.clone());
354        let app_data = match parent {
355            Some(ref parent) => parent.parent_root.app_data.clone(),
356            None => Rc::default(),
357        };
358        let subtree = Rc::new(SubtreeData {
359            parent,
360            app_data,
361
362            subtree_id: tree_root_id,
363            host: host_element.clone(),
364            event_registry: RefCell::new(event_registry),
365            global: RefCell::new(host_handlers),
366        });
367        subtree.app_data.borrow_mut().add_subtree(&subtree);
368        subtree
369    }

379    // Bubble a potential parent until it reaches an internal element
380    fn bubble_to_inner_element(
381        &self,
382        parent_el: Element,
383        should_bubble: bool,
384    ) -> Option<(&Self, Element)> {
385        let mut next_subtree = self;
386        let mut next_el = parent_el;
387        if !should_bubble && next_subtree.host.eq(&next_el) {
388            return None;
389        }
390        while next_subtree.host.eq(&next_el) {
391            // we've reached the host, delegate to a parent if one exists
392            let parent = next_subtree.parent.as_ref()?;
393            next_subtree = &parent.parent_root;
394            next_el = parent.mount_element.clone();
395        }
396        Some((next_subtree, next_el))
397    }
398
399    fn start_bubbling_if_responsible<'s>(
400        &'s self,
401        event: &'s Event,
402    ) -> Option<impl 's + Iterator<Item = (&'s SubtreeData, Element)>> {
403        // Note: the event is not necessarily indentically the same object for all installed
404        // handlers hence this cache can be unreliable. Hence the cached repsonsible_tree_id
405        // might be missing. On the other hand, due to event retargeting at shadow roots,
406        // the cache might be wrong! Keep in mind that we handle events in the capture
407        // phase, so top-down. When descending and retargeting into closed shadow-dom, the
408        // event might have been handled 'prematurely'. TODO: figure out how to prevent this
409        // and establish correct event handling for closed shadow root. Note: Other
410        // frameworks also get this wrong and dispatch such events multiple times.
411        let event_path = event.composed_path();
412        let derived_cached_key = event_path.length();
413        let cached_branding = if matches!(event.cache_key(), Some(cache_key) if cache_key == derived_cached_key)
414        {
415            event.subtree_id()
416        } else {
417            None
418        };
419        if matches!(cached_branding, Some(responsible_tree_id) if responsible_tree_id != self.subtree_id)
420        {
421            // some other handler has determined (via this function, but other `self`) a subtree
422            // that is responsible for handling this event, and it's not this subtree.
423            return None;
424        }
425        // We're tasked with finding the subtree that is reponsible with handling the event, and/or
426        // run the handling if that's `self`.
427        let target = event_path.get(0).dyn_into::<Element>().ok()?;
428        let should_bubble = BUBBLE_EVENTS.load(Ordering::Relaxed) && event.bubbles();
429        // We say that the most deeply nested subtree is "responsible" for handling the event.
430        let (responsible_tree_id, bubbling_start) = if let Some(branding) = cached_branding {
431            (branding, target.clone())
432        } else if let Some(branding) = find_closest_branded_element(target.clone(), should_bubble) {
433            let BrandingSearchResult {
434                branding,
435                closest_branded_ancestor,
436            } = branding;
437            event.set_subtree_id(branding);
438            event.set_cache_key(derived_cached_key);
439            (branding, closest_branded_ancestor)
440        } else {
441            // Possible only? if bubbling is disabled
442            // No tree should handle this event
443            event.set_subtree_id(NONE_TREE_ID);
444            event.set_cache_key(derived_cached_key);
445            return None;
446        };
447        if self.subtree_id != responsible_tree_id {
448            return None;
449        }
450        if self.host.eq(&target) {
451            // One more special case: don't handle events that get fired directly on a subtree host
452            return None;
453        }
454        Some(start_bubbling_from(self, bubbling_start, should_bubble))
455        // # More details: When nesting occurs
456        //
457        // Event listeners are installed only on the subtree roots. Still, those roots can
458        // nest. This could lead to events getting handled multiple times. We want event handling to
459        // start at the most deeply nested subtree.
460        //
461        // A nested subtree portals into an element that is controlled by the user and rendered
462        // with VNode::VRef. We get the following dom nesting:
463        //
464        // AppRoot > .. > UserControlledVRef > .. > NestedTree(PortalExit) > ..
465        // --------------                          ----------------------------
466        // The underlined parts of the hierarchy are controlled by Yew.
467        //
468        // from the following virtual_dom
469        // <AppRoot>
470        //   {VNode::VRef(<div><div id="portal_target" /></div>)}
471        //   {create_portal(<NestedTree />, #portal_target)}
472        // </AppRoot>
473    }
474
475    /// Handle a global event firing
476    fn handle(&self, desc: EventDescriptor, event: Event) {
477        let run_handler = |root: &Self, el: &Element| {
478            let handler = Registry::get_handler(root.event_registry(), el, &desc);
479            if let Some(handler) = handler {
480                handler(&event)
481            }
482        };
483        if let Some(bubbling_it) = self.start_bubbling_if_responsible(&event) {
484            test_log!("Running handler on subtree {}", self.subtree_id);
485            for (subtree, el) in bubbling_it {
486                if event.cancel_bubble() {
487                    break;
488                }
489                run_handler(subtree, &el);
490            }
491        }
492    }
493
494    fn add_listener(self: &Rc<Self>, desc: &EventDescriptor) {
495        let this = self.clone();
496        let listener = {
497            let desc = desc.clone();
498            move |e: &Event| {
499                this.handle(desc.clone(), e.clone());
500            }
501        };
502        self.host_handlers()
503            .borrow_mut()
504            .add_listener(desc, listener);
505    }
506}
