9impl WorkerFnType for OneshotFn {
10    type OutputType = Type;
11    type RecvType = Type;
12
13    fn attr_name() -> &'static str {
14        "oneshot"
15    }
16
17    fn worker_type_name() -> &'static str {
18        "oneshot"
19    }
20
21    fn parse_recv_type(sig: &Signature) -> syn::Result<Self::RecvType> {
22        let mut inputs = sig.inputs.iter();
23        let arg = inputs
24            .next()
25            .ok_or_else(|| syn::Error::new_spanned(&sig.ident, "expected 1 argument"))?;
26
27        let ty = Self::extract_fn_arg_type(arg)?;
28
29        Self::assert_no_left_argument(inputs, 1)?;
30
31        Ok(ty)
32    }
33
34    fn parse_output_type(sig: &Signature) -> syn::Result<Self::OutputType> {
35        let ty = match &sig.output {
36            ReturnType::Default => {
37                parse_quote! { () }
38            }
39            ReturnType::Type(_, ty) => *ty.clone(),
40        };
41
42        Ok(ty)
43    }
44}

46pub fn oneshot_impl(
47    name: WorkerName,
48    mut worker_fn: WorkerFn<OneshotFn>,
49) -> syn::Result<TokenStream> {
50    worker_fn.merge_worker_name(name)?;
51
52    let struct_attrs = worker_fn.filter_attrs_for_worker_struct();
53    let oneshot_impl_attrs = worker_fn.filter_attrs_for_worker_impl();
54    let phantom_generics = worker_fn.phantom_generics();
55    let oneshot_name = worker_fn.worker_name();
56    let fn_name = worker_fn.inner_fn_ident();
57    let inner_fn = worker_fn.print_inner_fn();
58
59    let WorkerFn {
60        recv_type: input_type,
61        generics,
62        output_type,
63        vis,
64        is_async,
65        ..
66    } = worker_fn;
67    let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
68    let fn_generics = ty_generics.as_turbofish();
69
70    let in_ident = Ident::new("_input", Span::mixed_site());
71
72    let fn_call = if is_async {
73        quote! { #fn_name #fn_generics (#in_ident).await }
74    } else {
75        quote! { #fn_name #fn_generics (#in_ident) }
76    };
77    let crate_name = WorkerFn::<OneshotFn>::worker_crate_name();
78
79    let quoted = quote! {
80        #(#struct_attrs)*
81        #[allow(unused_parens)]
82        #vis struct #oneshot_name #generics #where_clause {
83            inner: ::std::pin::Pin<::std::boxed::Box<dyn ::std::future::Future<Output = #output_type>>>,
84            _marker: ::std::marker::PhantomData<(#phantom_generics)>,
85        }
86
87        // we cannot disable any lints here because it will be applied to the function body
88        // as well.
89        #(#oneshot_impl_attrs)*
90        impl #impl_generics ::#crate_name::oneshot::Oneshot for #oneshot_name #ty_generics #where_clause {
91            type Input = #input_type;
92
93            fn create(#in_ident: Self::Input) -> Self {
94                #inner_fn
95
96                Self {
97                    inner: ::std::boxed::Box::pin(
98                        async move {
99                            #fn_call
100                        }
101                    ),
102                    _marker: ::std::marker::PhantomData,
103                }
104            }
105        }
106
107        impl #impl_generics ::std::future::Future for #oneshot_name #ty_generics #where_clause {
108            type Output = #output_type;
109
110            fn poll(mut self: ::std::pin::Pin<&mut Self>, cx: &mut ::std::task::Context<'_>) -> ::std::task::Poll<Self::Output> {
111                ::std::future::Future::poll(::std::pin::Pin::new(&mut self.inner), cx)
112            }
113        }
114
115        impl #impl_generics ::#crate_name::Registrable for #oneshot_name #ty_generics #where_clause {
116            type Registrar = ::#crate_name::oneshot::OneshotRegistrar<Self>;
117
118            fn registrar() -> Self::Registrar {
119                ::#crate_name::oneshot::OneshotRegistrar::<Self>::new()
120            }
121        }
122
123        impl #impl_generics ::#crate_name::Spawnable for #oneshot_name #ty_generics #where_clause {
124            type Spawner = ::#crate_name::oneshot::OneshotSpawner<Self>;
125
126            fn spawner() -> Self::Spawner {
127                ::#crate_name::oneshot::OneshotSpawner::<Self>::new()
128            }
129        }
130    };
131
132    Ok(quoted)
133}
